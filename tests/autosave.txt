# Autosave + restore: idle 3 s, 20 s apart, only stopped; boot restores the working song, slot and dirty flag
wait 100
expect autosaves 0
tap GLO
turn K1 3
expect slot 3
tap HOME
tapkey 7
tapkey 9
expect notes 2
expect dirty 1
wait 2000
expect autosaves 0
wait 1500
expect autosaves 1
# rate limit: a second edit right after waits for the 20 s since the last write
tapkey 11
expect notes 3
wait 5000
expect autosaves 1
wait 16000
expect autosaves 2
# power cycle: the working song comes back with its unsaved edits, dirty flag and slot
reboot
wait 100
expect notes 3
expect dirty 1
expect slot 3
# never while playing: edit, play, still nothing; stopped, written after the idle time
